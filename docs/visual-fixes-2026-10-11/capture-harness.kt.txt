    private fun shell(command: String) = ParcelFileDescriptor.AutoCloseInputStream(InstrumentationRegistry.getInstrumentation().uiAutomation.executeShellCommand(command)).use { String(it.readBytes()).trim() }
    private fun command(path: String, body: String = "{}") {
        (URL("http://127.0.0.1:18080/test/$path").openConnection() as HttpURLConnection).apply {
            requestMethod = "POST"; doOutput = true; outputStream.use { it.write(body.toByteArray()) }; inputStream.close(); disconnect()
        }
    }
    private fun capture(name: String) {
        compose.waitForIdle()
        InstrumentationRegistry.getInstrumentation().uiAutomation.waitForIdle(500,5000)
        android.os.SystemClock.sleep(250)
        shell("mkdir -p /data/local/tmp/mobivious-visual-audit")
        shell("screencap -p /data/local/tmp/mobivious-visual-audit/$name.png")
        android.util.Log.i("VisualAudit", "CAPTURE $name")
    }
    private fun reacquire() = ui { ActivityLifecycleMonitorRegistry.getInstance().getActivitiesInStage(Stage.RESUMED).filterIsInstance<MainActivity>().firstOrNull()?.let { activity = it } }
    private fun profile(narrow: Boolean, dark: Boolean) {
        shell("wm density ${if (narrow) "672" else "480"}")
        shell("settings put system font_scale ${if (narrow) "2.0" else "1.0"}")
        compose.waitForIdle(); reacquire()
        command("preferences", JSONObject().put("dark_mode",if(dark) "dark" else "light").toString())
        ui { if(vm.account.value==null) vm.store.guestDeArrow(AccountPreferences(autoplay=false,darkMode=if(dark) "dark" else "light")); vm.refreshSharedSettings() }
        until { vm.preferences.value.darkMode == if(dark) "dark" else "light" }
    }
    private fun back() { ui { activity.onBackPressedDispatcher.onBackPressed() }; compose.waitForIdle() }
    private fun login() {
        ui { vm.action { vm.login("AuditViewer", "transient-password") } }
        until { vm.account.value != null && !vm.accountBusy.value && vm.subscriptionChannels.value.loaded }
    }
    private fun settings() {
        ui { vm.navigate("You") }; compose.onNodeWithTag("global-settings").performClick()
    }
    private fun page(name: String) {
        compose.onNodeWithTag("settings-root").performScrollToNode(hasText(name))
        compose.onNodeWithText(name, substring = false).performClick(); compose.waitForIdle()
    }

        login()
        for ((narrow,dark) in listOf(false to false, false to true, true to true)) {

                page(name); capture("settings-${name.replace(' ','-')}-$suffix-0")
                if (!dark || narrow) {
                    val count=when(name){"Playback"->5;"AI channel filter"->4;"DeArrow"->3;else->2}

    @Test fun launcherMaskPreviews() {
        val i=InstrumentationRegistry.getInstrumentation()
        val drawable=i.targetContext.packageManager.getApplicationIcon(i.targetContext.packageName) as AdaptiveIconDrawable
        val dir=File(i.targetContext.cacheDir,"visual-audit-icons");dir.mkdirs()
        for(shape in listOf("circle","square","rounded-square")) {
            val bitmap=Bitmap.createBitmap(432,432,Bitmap.Config.ARGB_8888);val canvas=Canvas(bitmap)
            val path=Path().apply{when(shape){"circle"->addCircle(216f,216f,216f,Path.Direction.CW);"square"->addRect(0f,0f,432f,432f,Path.Direction.CW);else->addRoundRect(0f,0f,432f,432f,70f,70f,Path.Direction.CW)}}
            canvas.clipPath(path)
            // AdaptiveIconDrawable extends each layer by one quarter of the masked viewport.
            drawable.background.setBounds(-108,-108,540,540);drawable.foreground.setBounds(-108,-108,540,540)
            drawable.background.draw(canvas);drawable.foreground.draw(canvas)
            File(dir,"icon-$shape.png").outputStream().use{bitmap.compress(Bitmap.CompressFormat.PNG,100,it)}
        }
    }
